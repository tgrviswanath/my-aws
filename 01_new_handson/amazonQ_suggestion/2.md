PROJECT 1: Basic VPC with Public Subnet - Complete Implementation Guide
1. Project Overview
Project Title
Basic VPC with Public Subnet - Foundation Level AWS Networking Project

Business/Problem Statement
In modern cloud architecture, organizations need secure, isolated network environments to host their applications. This project addresses the fundamental requirement of creating a basic Virtual Private Cloud (VPC) with a public subnet that can host web-facing resources like web servers, application servers, or bastion hosts that need direct internet connectivity.

Real-world scenarios:

Hosting a simple web application
Creating a development/testing environment
Setting up a bastion host for secure access
Building the foundation for more complex multi-tier architectures
Learning Objectives
By completing this project, you will:

Understand VPC fundamentals and AWS networking concepts
Learn to create and configure VPC components using multiple methods
Master the relationship between VPCs, subnets, route tables, and internet gateways
Gain hands-on experience with AWS Console, CLI, and Terraform
Understand security groups and network ACLs
Learn infrastructure as code principles
Develop skills in resource verification and troubleshooting
2. Architecture & Concepts
Core AWS Services Involved
Amazon VPC (Virtual Private Cloud) - Isolated network environment
Subnet - Network segment within the VPC
Internet Gateway (IGW) - Gateway for internet connectivity
Route Table - Traffic routing rules
Security Group - Instance-level firewall
Network ACL - Subnet-level firewall (default)
Service Interaction Flow
Internet → Internet Gateway → Route Table → Public Subnet → EC2 Instance
                                ↓
                         Security Group (Firewall)

High-Level Architecture Explanation
The architecture consists of:

VPC: A logically isolated network (10.0.0.0/16 CIDR block)
Public Subnet: A subnet (10.0.1.0/24) with internet access
Internet Gateway: Enables bidirectional internet communication
Route Table: Directs traffic between subnet and internet gateway
Security Group: Controls inbound/outbound traffic at instance level
Best Practices Followed
Use private IP ranges (RFC 1918)
Implement least privilege security
Enable DNS resolution and hostnames
Use descriptive naming conventions
Tag all resources for management
Follow AWS Well-Architected Framework principles

providers.tf
terraform {
  required_version = ">= 1.5"
  
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }

provider "aws" {
  region = var.aws_region
  
  default_tags {
    tags = {
      Project     = var.project_name
      Environment = var.environment
      ManagedBy   = "Terraform"
      Owner       = var.owner
    }
  }
}

variables.tf
variable "aws_region" {
  description = "AWS region for resources"
  type        = string
  default     = "us-east-1"
}

variable "project_name" {
  description = "Name of the project"
  type        = string
  default     = "VPC-Basic-Implementation"
}

variable "environment" {
  description = "Environment name"
  type        = string
  default     = "Development"
}

variable "owner" {
  description = "Owner of the resources"
  type        = string
  default     = "DevOps-Team"
}

variable "vpc_cidr" {
  description = "CIDR block for VPC"
  type        = string
  default     = "10.0.0.0/16"
  
  validation {
    condition     = can(cidrhost(var.vpc_cidr, 0))
    error_message = "VPC CIDR must be a valid IPv4 CIDR block."
  }
}

variable "public_subnet_cidr" {
  description = "CIDR block for public subnet"
  type        = string
  default     = "10.0.1.0/24"
  
  validation {
    condition     = can(cidrhost(var.public_subnet_cidr, 0))
    error_message = "Public subnet CIDR must be a valid IPv4 CIDR block."
  }
}

variable "availability_zone" {
  description = "Availability zone for public subnet"
  type        = string
  default     = "us-east-1a"
}

variable "enable_dns_hostnames" {
  description = "Enable DNS hostnames in VPC"
  type        = bool
  default     = true
}

variable "enable_dns_support" {
  description = "Enable DNS support in VPC"
  type        = bool
  default     = true
}

main.tf
# Data source to get available AZs
data "aws_availability_zones" "available" {
  state = "available"
}

# Data source to get current public IP
data "http" "myip" {
  url = "https://checkip.amazonaws.com"
}

locals {
  my_ip = chomp(data.http.myip.response_body)
}

# VPC
resource "aws_vpc" "main" {
  cidr_block           = var.vpc_cidr
  enable_dns_hostnames = var.enable_dns_hostnames
  enable_dns_support   = var.enable_dns_support

  tags = {
    Name = "${var.project_name}-vpc"
    Type = "Main VPC"
  }
}

# Internet Gateway
resource "aws_internet_gateway" "main" {
  vpc_id = aws_vpc.main.id

  tags = {
    Name = "${var.project_name}-igw"
    Type = "Internet Gateway"
  }
}

# Public Subnet
resource "aws_subnet" "public" {
  vpc_id                  = aws_vpc.main.id
  cidr_block              = var.public_subnet_cidr
  availability_zone       = var.availability_zone
  map_public_ip_on_launch = true

  tags = {
    Name = "${var.project_name}-public-subnet"
    Type = "Public Subnet"
  }
}

# Route Table for Public Subnet
resource "aws_route_table" "public" {
  vpc_id = aws_vpc.main.id

  route {
    cidr_block = "0.0.0.0/0"
    gateway_id = aws_internet_gateway.main.id
  }

  tags = {
    Name = "${var.project_name}-public-rt"
    Type = "Public Route Table"
  }
}

# Route Table Association
resource "aws_route_table_association" "public" {
  subnet_id      = aws_subnet.public.id
  route_table_id = aws_route_table.public.id
}

# Security Group for Web Servers
resource "aws_security_group" "web" {
  name        = "${var.project_name}-web-sg"
  description = "Security group for web servers in public subnet"
  vpc_id      = aws_vpc.main.id

  # HTTP access from anywhere
  ingress {
    description = "HTTP"
    from_port   = 80
    to_port     = 80
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  # HTTPS access from anywhere
  ingress {
    description = "HTTPS"
    from_port   = 443
    to_port     = 443
    protocol    = "tcp"
    cidr_blocks = ["0.0.0.0/0"]
  }

  # SSH access from your IP only
  ingress {
    description = "SSH"
    from_port   = 22
    to_port     = 22
    protocol    = "tcp"

tags: Metadata for resource identification and management

depends_on = [aws_route_table_association.public]

8. Observations & Learning Notes
What to Observe During Execution
1. Resource Creation Order
Observation: Notice the order in which AWS creates resources:

VPC (foundation)
Internet Gateway (can be created independently)
Subnet (requires VPC)
Route Table (requires VPC)
Security Group (requires VPC)
Route Table Association (requires both route table and subnet)
Learning: AWS enforces dependency relationships automatically in Terraform, but you must manage them manually in CLI.

2. State Transitions
Observation: Watch resource states change:

VPC: pending → available
Subnet: pending → available
Internet Gateway: detached → attaching → attached
Learning: Always wait for resources to reach available state before proceeding.

3. Automatic Resource Creation
Observation: When you create a VPC, AWS automatically creates:

Default Route Table
Default Network ACL
Default Security Group
Learning: These default resources provide basic connectivity but should be customized for production use.

Internal AWS Behavior
1. IP Address Allocation
What happens: When you specify CIDR blocks:

AWS reserves first 4 and last 1 IP addresses in each subnet
Example: In 10.0.1.0/24 subnet:
10.0.1.0: Network address
10.0.1.1: VPC router
10.0.1.2: DNS server
10.0.1.3: Future use
10.0.1.255: Broadcast address
Learning: Always account for 5 reserved IPs when planning subnet sizes.

2. Route Table Behavior
What happens: AWS automatically:

Adds local routes for VPC CIDR blocks
Processes routes in order of specificity (most specific first)
Uses longest prefix matching for route selection
Learning: Local routes always take precedence over custom routes.

3. Security Group Processing
What happens: AWS processes security group rules:

Evaluates all rules (no order dependency)
Allows traffic if ANY rule matches
Automatically allows return traffic (stateful)
Denies everything not explicitly allowed
Learning: Security groups are "allow-only" - you cannot create deny rules.

Resource Dependencies Created
Dependency Chain
VPC
├── Internet Gateway (attached to VPC)
├── Subnet (created in VPC)
├── Route Table (created in VPC)
│   ├── Route (points to IGW)
│   └── Association (links to Subnet)
└── Security Group (created in VPC)

Critical Dependencies
Internet Gateway → VPC: IGW must be attached to VPC for internet access
Route Table → Internet Gateway: Route must reference IGW for internet routing
Subnet → Route Table: Association required for routing rules to apply
Security Group → VPC: Security groups are VPC-specific

Complete Cleanup Script
#!/bin/bash
# cleanup.sh - Complete resource cleanup

set -e  # Exit on any error

echo "=== STARTING CLEANUP PROCESS ==="

# Get resource IDs
VPC_ID=$(aws ec2 describe-vpcs --filters "Name=tag:Name,Values=project-vpc-basic-vpc" --query 'Vpcs[0].VpcId' --output text 2>/dev/null || echo "None")
SUBNET_ID=$(aws ec2 describe-subnets --filters "Name=tag:Name,Values=project-vpc-basic-public-subnet" --query 'Subnets[0].SubnetId' --output text 2>/dev/null || echo "None")
IGW_ID=$(aws ec2 describe-internet-gateways --filters "Name=tag:Name,Values=project-vpc-basic-igw" --query 'InternetGateways[0].InternetGatewayId' --output text 2>/dev/null || echo "None")
RT_ID=$(aws ec2 describe-route-tables --filters "Name=tag:Name,Values=project-vpc-basic-public-rt" --query 'RouteTables[0].RouteTableId' --output text 2>/dev/null || echo "None")
SG_ID=$(aws ec2 describe-security-groups --filters "Name=group-name,Values=project-vpc-basic-web-sg" --query 'SecurityGroups[0].GroupId' --output text 2>/dev/null || echo "None")

echo "Found resources:"
echo "VPC: $VPC_ID"
echo "Subnet: $SUBNET_ID"
echo "IGW: $IGW_ID"
echo "Route Table: $RT_ID"
echo "Security Group: $SG_ID"
echo

# Function to check if resource exists
resource_exists() {
    [ "$1" != "None" ] && [ "$1" != "" ]
}

# 1. Delete Security Group
if resource_exists "$SG_ID"; then
    echo "Deleting Security Group: $SG_ID"
    aws ec2 delete-security-group --group-id $SG_ID
    echo "✓ Security Group deleted"
else
    echo "Security Group not found or already deleted"
fi

# 2. Disassociate and Delete Route Table
if resource_exists "$RT_ID"; then
    echo "Disassociating Route Table: $RT_ID"
    # Get association ID
    ASSOC_ID=$(aws ec2 describe-route-tables --route-table-ids $RT_ID --query 'RouteTables[0].Associations[?SubnetId!=null].RouteTableAssociationId' --output text 2>/dev/null || echo "None")
    
    if resource_exists "$ASSOC_ID"; then
        aws ec2 disassociate-route-table --association-id $ASSOC_ID
        echo "✓ Route Table disassociated"
    fi
    
    echo "Deleting Route Table: $RT_ID"
    aws ec2 delete-route-table --route-table-id $RT_ID
    echo "✓ Route Table deleted"
else
    echo "Route Table not found or already deleted"
fi

# 3. Delete Subnet
if resource_exists "$SUBNET_ID"; then
    echo "Deleting Subnet: $SUBNET_ID"
    aws ec2 delete-subnet --subnet-id $SUBNET_ID
    echo "✓ Subnet deleted"
else
    echo "Subnet not found or already deleted"
fi

# 4. Detach and Delete Internet Gateway
if resource_exists "$IGW_ID"; then
    echo "Detaching Internet Gateway: $IGW_ID"
    aws ec2 detach-internet-gateway --internet-gateway-id $IGW_ID --vpc-id $VPC_ID 2>/dev/null || echo "IGW already detached"
    
    echo "Deleting Internet Gateway: $IGW_ID"
    aws ec2 delete-internet-gateway --internet-gateway-id $IGW_ID
    echo "✓ Internet Gateway deleted"
else
    echo "Internet Gateway not found or already deleted"
fi

# 5. Delete VPC
if resource_exists "$VPC_ID"; then
    echo "Deleting VPC: $VPC_ID"
    aws ec2 delete-vpc --vpc-id $VPC_ID
    echo "✓ VPC deleted"
else
    echo "VPC not found or already deleted"
fi

echo
echo "=== CLEANUP COMPLETE ==="
echo "All resources have been successfully deleted."
